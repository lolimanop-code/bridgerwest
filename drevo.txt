local VirtualInputManager = game:GetService("VirtualInputManager")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService('TweenService')

local teleport_table = {
    location1 = Vector3.new(-1482.459716796875, 84, -3800.93994140625),
    location2 = Vector3.new(-1509.91455078125, 85, -3832.979736328125),
    location3 = Vector3.new(-1490.33740234375, 83.5, -3890.40625),
    location4 = Vector3.new(-1526.2386474609375, 85, -3926.34228515625),
    location5 = Vector3.new(-1514.849853515625, 78.5, -3995.408935546875)
}

local FishingRodName = "LumberAxe"
local TOGGLE_KEY = Enum.KeyCode.L

local isRunning = false
local loopTask = nil

local function SimulateMouseClick(x, y)
    VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 1)
    task.wait(0.05)
    VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 1)
end

local function bypass_teleport(v)
    local character = LocalPlayer.Character
    if not character then return end
    
    local hrp = character:FindFirstChild('HumanoidRootPart')
    if not hrp then return end
    
    local distance = (hrp.Position - v).Magnitude
    local sec = math.max(distance / 35, 0.1)  -- минимум 0.1 сек
    
    local tweenInfo = TweenInfo.new(sec, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.new(v)})
    tween:Play()
    tween.Completed:Wait()
end

local function updateReferences()
    local character = LocalPlayer.Character
    if not character then return false end
    return character:FindFirstChild("Humanoid") ~= nil
end

local function equipAxe()
    local character = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    
    if not character or not backpack then
        print("[⏳] Персонаж или инвентарь не загружены")
        return false
    end
    
    local humanoid = character:FindFirstChild("Humanoid")
    if not humanoid then return false end
    
    local rod = backpack:FindFirstChild(FishingRodName)
    if rod then
        humanoid:EquipTool(rod)
        print("[✅] Инструмент '" .. FishingRodName .. "' в руке!")
        return true
    else
        print("[❌] Инструмент '" .. FishingRodName .. "' не найден")
        return false
    end
end

local function startLoop()
    if isRunning then return end
    isRunning = true
    
    loopTask = task.spawn(function()
        while isRunning do
            for _, v in pairs(teleport_table) do
                if not isRunning then break end
                bypass_teleport(v)
                
                for _ = 1, 150 do  -- 15/0.1 = 150
                    if not isRunning then break end
                    SimulateMouseClick(500, 500)
                    task.wait(0.1)
                end
            end
        end
    end)
end

local function stopLoop()
    isRunning = false
    loopTask = nil
end

local function toggleLoop()
    if isRunning then
        stopLoop()
    else
        startLoop()
        equipAxe()
    end
end

-- Переподключение при появлении персонажа
LocalPlayer.CharacterAdded:Connect(function(character)
    print("[🔄] Персонаж появился")
    character:WaitForChild("Humanoid")
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == TOGGLE_KEY then
        toggleLoop()
    end
end)
